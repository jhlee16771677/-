    function drawPolygonArrow(ctx, fx, fy, tx, ty, color) {
      const a = Math.atan2(ty-fy, tx-fx);
      const len = Math.hypot(tx-fx, ty-fy);
      const headW = 16, headL = 16, shaftW = 8;
      ctx.save(); ctx.translate(fx, fy); ctx.rotate(a);
      ctx.fillStyle = color;
      ctx.beginPath();
      ctx.moveTo(0, -shaftW/2); ctx.lineTo(len - headL, -shaftW/2);
      ctx.lineTo(len - headL, -headW/2); ctx.lineTo(len, 0);
      ctx.lineTo(len - headL, headW/2); ctx.lineTo(len - headL, shaftW/2);
      ctx.lineTo(0, shaftW/2); ctx.closePath();
      ctx.shadowColor = 'rgba(0,0,0,0.2)'; ctx.shadowBlur = 4; ctx.shadowOffsetY = 2;
      ctx.fill(); ctx.restore();
    }

    }

    function updateParticles(ctx, isTab4) {
      if(particles.length < 25 && Math.random() < 0.07) spawnParticle(0);
      
      for(let i = particles.length-1; i>=0; i--) {
        let p = particles[i];
        let px = 0, py = 0;
        let v = p.baseSpeed * simSpeed;
        
        if(!isPlaying) v = 0;
        if(isTab4) v *= 0.85; // 3번 탭에서만 속도를 아주 조금 더 느리게 조절

        let rules = isTab4 ? customRules[p.type] : defaultRules[p.type];

        if(p.route === 'artery') {
          let L = 60; p.t += v / L;
          px = 40 + p.t * L; py = 80 + p.laneT;
          if(p.t >= 1) { p.t = 0; p.route = 'artery_curve'; }
        }
        else if(p.route === 'artery_curve') {
          let L = 55; p.t += v / L;
          px = 100 + p.t * 30; py = 80 + p.t * 30 + p.laneT; 
          if(p.t >= 1) {
            p.t = 0;
            let willFilter = rules.filter;
            if (willFilter) {
                if (p.type === 'urea') willFilter = Math.random() < 0.7;
                /* 물은 토리에서 모두 여과되도록 이전의 잘못된 확률 코드를 제거했습니다. 
                   (이후 세뇨관에서 85%는 재흡수되고 나머지는 오줌으로 배출됩니다.) */
            }
            p.route = willFilter ? 'tubule_v' : 'efferent_curve';
          }
        }
        else if(p.route === 'efferent_curve') {
          let L = 55; p.t += v / L;
          px = 170 + p.t * 30; py = 110 - p.t * 30 + p.laneT;
          if(p.t >= 1) { p.t = 0; p.route = 'efferent_h'; }
        }
        else if(p.route === 'efferent_h') {
          let L = 80; p.t += v / L;
          px = 200 + p.t * L; py = 80 + p.laneT;
          if(p.t >= 1) { p.t = 0; p.route = 'efferent_v'; }
        }
        else if(p.route === 'efferent_v') {
          let L = 180; p.t += v / L;
          px = 280 + p.laneT; py = 80 + p.t * L;
          if(p.t >= 1) { p.t = 0; p.route = 'cap_main'; }
        }
        else if(p.route === 'tubule_v') {
          let L = 175; p.t += v / L;
          px = 150 + p.laneT; py = 115 + p.t * L;
          if(p.t >= 1) { p.t = 0; p.route = 'tubule_c'; }
        }
        else if(p.route === 'tubule_c') {
          let L = 63; p.t += v / L;
          let angle = Math.PI - p.t * (Math.PI/2); 
          let r = 40 - p.laneT;
          px = 190 + r * Math.cos(angle); py = 290 + r * Math.sin(angle);
          if(p.t >= 1) { p.t = 0; p.route = 'tubule_h'; }
        }
        else if(p.route === 'tubule_h') {
          let L = 760; p.t += v / L;
          px = 190 + p.t * L; py = 330 + p.laneT;

          if(!p.reabsorbed && p.t > 0.3 && p.t < 0.35) { 
            let doReab = rules.reabsorb;
            
            // 재흡수(O) 상태일 때: 물방울이 오줌으로 직진하는 빈도를 살짝 높임 (약 75% 재흡수, 25% 오줌으로)
            if(doReab && p.type === 'water') {
                doReab = Math.random() < 0.75; 
            }
            
            if(doReab) { 
               p.reabsorbed = true; p.route = 'reab_up'; 
               p.t = 0; p.sX = px; p.sY = py; 
            } else {
               // 오류 수정: 재흡수되지 않기로 결정되면, 다음 프레임에서 또 계산하지 않도록 'skipped' 처리
               p.reabsorbed = 'skipped';
            }
          }
          if(p.t >= 1) p.done = true;
        }
        else if(p.route === 'cap_main') {
          let L = 455; p.t += v / L;
          px = 280 + p.t * L; py = 260 + p.laneT;

          if(!p.secreted && p.t > 0.65 && p.t < 0.70) {
            let doSec = rules.secrete;
            if (doSec && p.type === 'urea') doSec = Math.random() < 0.95;
            if (doSec) {
                p.secreted = true; p.route = 'sec_down';
                p.t = 0; p.sX = px; p.sY = py; 
            } else {
                // 오류 수정: 분비되지 않기로 결정되면, 다음 프레임에서 또 계산하지 않도록 'skipped' 처리
                p.secreted = 'skipped';
            }
          }
          if(p.t >= 1) { p.t = 0; p.route = 'vein_down'; }
        }
        else if(p.route === 'reab_up') {
          let L = 70; p.t += v / L;
          px = p.sX; py = p.sY - p.t * L; 
          if(p.t >= 1) { 
             p.route = 'cap_main'; 
             p.t = (px - 280) / 455; 
          }
        }
        else if(p.route === 'sec_down') {
          let L = 70; p.t += v / L;
          px = p.sX; py = p.sY + p.t * L; 
          if(p.t >= 1) { 
             p.route = 'tubule_h'; 
             p.t = (px - 190) / 760; 
          }
        }
        else if(p.route === 'vein_down') {
          let L = 190; p.t += v / L;
          px = 735 + p.laneT; py = 260 + p.t * L; 
          if(p.t >= 1) p.done = true;
        }

        let match = (currentFlt === 'all') || 
                    (currentFlt === 'rbc' && (p.type==='rbc'||p.type==='prot')) || 
                    (currentFlt === 'glu' && p.type==='gluc') || 
                    (currentFlt === 'urea' && p.type==='urea') || 
                    (currentFlt === 'water' && p.type==='water');
        
        ctx.globalAlpha = match ? 1.0 : 0.15;
        // p.t가 음수일 때는 화면 왼쪽 바깥에서부터 천천히 진입하도록 처리됨
        if(px > 0 && py > 0) drawP(ctx, p, px, py);
        ctx.globalAlpha = 1.0;
        
        if(p.done || py > 480 || px > 980) particles.splice(i, 1);
      }
    }

    function drawP(ctx, p, x, y) {
      if(p.type === 'rbc') { 
        ctx.beginPath(); ctx.arc(x, y, 9, 0, Math.PI*2); ctx.fillStyle = '#fca5a5'; ctx.fill();
        ctx.lineWidth = 2.5; ctx.strokeStyle = '#ef4444'; ctx.stroke();
      } else if (p.type === 'prot') { 
        ctx.fillStyle = '#15803d'; ctx.beginPath(); 
        for(let i=0; i<5; i++) { const a = (i * 2 * Math.PI / 5) - Math.PI/2; const px = x + 11 * Math.cos(a), py = y + 11 * Math.sin(a); if(i===0) ctx.moveTo(px, py); else ctx.lineTo(px, py); }
        ctx.closePath(); ctx.fill();
      } else if (p.type === 'gluc') { 
        ctx.fillStyle = '#a855f7'; ctx.beginPath(); 
        for(let i=0; i<6; i++) { const a = (i * 2 * Math.PI / 6); const px = x + 5.5 * Math.cos(a), py = y + 5.5 * Math.sin(a); if(i===0) ctx.moveTo(px, py); else ctx.lineTo(px, py); }
        ctx.closePath(); ctx.fill();
      } else if (p.type === 'urea') { 
        ctx.fillStyle = '#fde047'; ctx.strokeStyle = '#eab308'; ctx.lineWidth = 1.5;
        ctx.fillRect(x-4, y-4, 8, 8); ctx.strokeRect(x-4, y-4, 8, 8);
      } else if (p.type === 'water') { 
        ctx.fillStyle = '#38bdf8'; ctx.beginPath(); 
        ctx.moveTo(x, y - 7); ctx.bezierCurveTo(x+5, y-1, x+5, y+5, x, y+7); ctx.bezierCurveTo(x-5, y+5, x-5, y-1, x, y-7);
        ctx.closePath(); ctx.fill();
      }
    }

    function allowDrop(ev) { ev.preventDefault(); }
    function drag(ev) { ev.dataTransfer.setData("text", ev.target.id); }
    function drop(ev) { ev.preventDefault(); dipTestStrip(); }

    function dipTestStrip() {
      if(patientRecords[selectedUrinaryPatient].status !== 'pending') return; 
      patientRecords[selectedUrinaryPatient].status = 'testing';
      showToast('반응 완료! 모둠원과 상의하여 1, 2, 3단계 답안을 기록하세요.', '🧪');
      renderDiagnosticReport();
    }

    function saveCurrentPatient() {
      const subEl = document.querySelector('input[name="ur-substance"]:checked');
      const causeEl = document.querySelector('input[name="ur-cause"]:checked');
      const trtEl = document.querySelector('input[name="ur-trt"]:checked');

      if (!subEl || !causeEl || !trtEl) {
         showToast('1단계, 2단계, 3단계의 답을 모두 선택해야 저장할 수 있습니다!', '⚠️');
         return false; 
      }

      patientRecords[selectedUrinaryPatient].answers = {
         sub: subEl.value, cause: causeEl.value, trt: trtEl.value
      };
      
      if(patientRecords[selectedUrinaryPatient].status === 'testing' || patientRecords[selectedUrinaryPatient].status === 'incorrect') {
         patientRecords[selectedUrinaryPatient].status = 'saved';
      }
      
      showToast('진단 기록이 임시 저장되었습니다.', '💾');
      updateBadges();
      renderDiagnosticReport(); 
      return true;
    }

    function saveCurrentPatientAndGrade() {
      if(saveCurrentPatient()) gradeAll();
    }

    function gradeAll() {
      let allCorrect = true;
      [1, 2, 3].forEach(i => {
        const rec = patientRecords[i];
        const pt = URINARY_PATIENTS[i];
        
        if (rec.answers && rec.answers.sub) {
          const isCorrect = (rec.answers.sub === pt.substance && rec.answers.cause === '1' && rec.answers.trt === '1');
          if (isCorrect) rec.status = 'correct';
          else { rec.status = 'incorrect'; allCorrect = false; }
        } else {
          allCorrect = false; 
        }
      });
      
      updateBadges();
      renderDiagnosticReport();

      if (allCorrect) setTimeout(showModal, 600);
      else showToast('진단이 틀린 환자가 있습니다. 오진(빨간색) 뱃지를 확인하고 다시 토의해보세요!', '❌');
    }

    function updateBadges() {
      let correctCount = 0;
      [1, 2, 3].forEach(i => {
        const badge = document.getElementById(`pt-badge-${i}`);
        const stat = patientRecords[i].status;
        
        if (stat === 'pending' || stat === 'testing') {
          badge.classList.add('hidden');
        } else {
          badge.classList.remove('hidden');
          badge.classList.remove('bg-emerald-500', 'bg-slate-500', 'bg-rose-500');
          
          if (stat === 'saved') { badge.classList.add('bg-slate-500'); badge.innerText = '기록완료'; } 
          else if (stat === 'correct') { badge.classList.add('bg-emerald-500'); badge.innerText = '진료완료'; correctCount++; } 
          else if (stat === 'incorrect') { badge.classList.add('bg-rose-500'); badge.innerText = '재진단요망'; }
        }
      });
      document.getElementById('clinic-score').innerText = correctCount;
    }

    function resetUrinaryLab() {
      patientRecords = {
        1: { status: 'pending', answers: { sub: null, cause: null, trt: null } },
        2: { status: 'pending', answers: { sub: null, cause: null, trt: null } },
        3: { status: 'pending', answers: { sub: null, cause: null, trt: null } }
      };
      isGradingUnlocked = false;
      updateBadges();
      selectUrinaryPt(1);
      showToast('모의 진료실이 초기화되었습니다. 처음부터 다시 시작합니다.', '🔄');
    }
